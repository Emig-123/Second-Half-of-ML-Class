from itertools import combinations

def apriori(onehot, min_support=0.01, use_colnames=True):
    data = onehot.to_numpy(dtype=bool)
    cols = list(onehot.columns)
    n = len(onehot)
    freq = {}
    counts = data.sum(axis=0)
    previous = []
    for j, count in enumerate(counts):
        if count / n >= min_support:
            item = (j,)
            freq[item] = count / n
            previous.append(item)
    k = 2
    while previous:
        prevset = set(previous)
        candidates = set()
        for a,b in combinations(previous, 2):
            merged = tuple(sorted(set(a).union(b)))
            if len(merged)==k and all(t in prevset for t in combinations(merged, k-1)):
                candidates.add(merged)
        current = []
        for candidate in sorted(candidates):
            share = float(np.count_nonzero(np.all(data[:, candidate], axis=1)) / n)
            if share >= min_support:
                current.append(candidate)
                freq[candidate] = share
        previous = current
        k += 1
    result = pd.DataFrame([
        {'support':support, 'itemsets':frozenset(cols[j] for j in key) if use_colnames else frozenset(key)}
        for key,support in freq.items()
    ])
    return result

def association_rules(itemsets, metric='confidence', min_threshold=0.2, num_itemsets=None):
    lookup = {frozenset(row.itemsets): float(row.support) for row in itemsets.itertuples()}
    rows=[]
    for union, union_support in lookup.items():
        if len(union)<2:continue
        for r in range(1,len(union)):
            for ant in combinations(sorted(union),r):
                ant=frozenset(ant)
                cons=union-ant
                if ant not in lookup or cons not in lookup:continue
                sa,sc=lookup[ant],lookup[cons]
                conf=union_support/sa
                lift=conf/sc
                row={'antecedents':ant,'consequents':cons,'antecedent support':sa,
                     'consequent support':sc,'support':union_support,'confidence':conf,
                     'lift':lift,'leverage':union_support-sa*sc,
                     'conviction':(1-sc)/(1-conf) if conf<1 else float('inf')}
                if row.get(metric,0)>=min_threshold:rows.append(row)
    return pd.DataFrame(rows)

make_rules = association_rules

Groceries data set¶

  • One month (30 days) of real point-of-sale transactions from a grocery store.
  • 9,835 transactions (baskets).
  • 169 product categories, such as whole milk, yogurt, bottled beer, etc.
  • Source: the data set Groceries in the R package arules, provided by Hahsler, Hornik, and Reutterer. The copy we use is a plain-text version hosted on GitHub.

The file format is different from the bakery file. The bakery file had one row per item bought. This file has one line per transaction, and each line is a comma-separated list of the items in that basket. Different baskets have different numbers of items.

The next cell reads each line into a single text column. (Normal CSV reading expects every row to have the same number of columns, so we read each line as one string, using a separator "|" that never occurs in the file.)

def support(items, onehot):
    if not items: return 1.0
    return float(onehot[list(items)].all(axis=1).mean())

def rule_metrics(antecedent, consequent, onehot):
    sa = support(antecedent, onehot)
    sc = support(consequent, onehot)
    joint = support(list(dict.fromkeys(antecedent + consequent)), onehot)
    confidence = joint / sa if sa else float('nan')
    lift = confidence / sc if sc else float('nan')
    return pd.Series({'support':joint, 'confidence':confidence, 'lift':lift})
