# Compatibility between versions of scikit-learn.
def one_hot():
    try:
        return OneHotEncoder(handle_unknown='ignore', sparse_output=False)
    except TypeError:
        return OneHotEncoder(handle_unknown='ignore', sparse=False)

def load_data(url, local_name, sep=','):
    """Prefer the original local file; otherwise use the instructor's GitHub URL."""
    from pathlib import Path
    p = Path('/mnt/data') / ('loans_full_schema.txt' if 'loans' in local_name else local_name)
    if p.exists():
        return pd.read_csv(p, sep=sep)
    try:
        import requests
        r = requests.get(url, timeout=12)
        r.raise_for_status()
        return pd.read_csv(io.BytesIO(r.content), sep=sep)
    except Exception as exc:
        print(f'DATA UNAVAILABLE: {local_name}: {type(exc).__name__}: {exc}')
        print(f'Upload {local_name} beside this notebook or rerun in Google Colab with internet.')
        return None

def rmse_cv(estimator, X, y, cv=CV):
    return -cross_val_score(estimator, X, y, scoring=SCORING, cv=cv, n_jobs=1).mean()

def preprocessing(frame, features, scale=True, ordinal=False):
    numeric = [c for c in features if pd.api.types.is_numeric_dtype(frame[c])]
    categorical = [c for c in features if c not in numeric]
    num_steps = [('impute', SimpleImputer(strategy='median'))]
    if scale:
        num_steps.append(('scale', StandardScaler()))
    cat_steps = [('impute', SimpleImputer(strategy='most_frequent'))]
    if ordinal:
        cat_steps.append(('encode', OrdinalEncoder(handle_unknown='use_encoded_value', unknown_value=-1)))
    else:
        cat_steps.append(('encode', one_hot()))
    return ColumnTransformer([
        ('num', Pipeline(num_steps), numeric),
        ('cat', Pipeline(cat_steps), categorical)
    ], remainder='drop')

def knn_pipeline(frame, features, k=10, metric='euclidean', weights='uniform', scale=True):
    return Pipeline([
        ('preprocess', preprocessing(frame, features, scale=scale)),
        ('model', KNeighborsRegressor(n_neighbors=k, metric=metric, weights=weights))
    ])

These preprocessing steps belong inside each Pipeline to avoid leakage.

Loans Data¶

The loans data set represents thousands of loans made through the Lending Club platform, which is a platform that allows individuals to lend to other individuals. Of course, not all loans are created equal. Someone who is a essentially a sure bet to pay back a loan will have an easier time getting a loan with a low interest rate than someone who appears to be riskier. And for people who are very risky? They may not even get a loan offer, or they may not have accepted the loan offer due to a high interest rate. It is important to keep that last part in mind, since this dataset only represents loans actually made, i.e. do not mistake this data for loan applications!

Some information about the data set is here. There are also some related exercises in this OpenIntro textbook.

Our goal will be to develop a regression model for predicting interest_rate.

# 6. Explore Lending Club data, fit multiple models, tune with GridSearchCV,
#    and report the lowest cross-validated interest_rate RMSE.
if df is not None:
    # Keep predictors available at origination, not repayment outcomes (data leakage).
    candidates = [
        'loan_amount', 'term', 'grade', 'sub_grade', 'annual_income', 'debt_to_income',
        'verified_income', 'homeownership', 'loan_purpose', 'purpose', 'state',
        'total_credit_limit', 'total_debit_limit', 'delinq_2y', 'inquiries_last_12m',
        'public_record_bankrupt', 'earliest_credit_line', 'num_accounts',
        'accounts_opened_24m', 'application_type'
    ]
    loan_features = [c for c in candidates if c in df.columns]
    loan_df = df.dropna(subset=['interest_rate']).copy()
    # Some loan inputs can be skewed: log1p of non-negative money amounts.
    for c in ['annual_income', 'loan_amount', 'total_credit_limit', 'total_debit_limit']:
        if c in loan_features:
            new_name = 'log_' + c
            loan_df[new_name] = np.log1p(pd.to_numeric(loan_df[c], errors='coerce').clip(lower=0))
            loan_features.remove(c)
            loan_features.append(new_name)
    # Exclude candidates with no recorded data.
    loan_features = [c for c in loan_features if loan_df[c].notna().any()]
    if not loan_features:
        raise ValueError('No recognized loan predictors were found. Inspect df.columns.')
    loan_df = loan_df.sample(n=min(2000, len(loan_df)), random_state=RANDOM_STATE).copy()
    loan_df = loan_df.sample(n=min(600, len(loan_df)), random_state=RANDOM_STATE).copy()
    loan_X = loan_df[loan_features].copy()
    loan_y = loan_df['interest_rate']
    print('Predictors:', loan_features)
    print('Missing fraction by predictor:')
    display(loan_X.isna().mean().sort_values(ascending=False).to_frame('Missing fraction'))
    fig, axes = plt.subplots(1, 2, figsize=(11, 4))
    loan_y.hist(ax=axes[0], bins=30)
    axes[0].set(title='Interest-rate distribution', xlabel='Interest rate', ylabel='Count')
    if 'grade' in loan_X.columns:
        loan_df.boxplot(column='interest_rate', by='grade', ax=axes[1])
        axes[1].set(title='Interest rate by loan grade', xlabel='Grade', ylabel='Interest rate')
        plt.suptitle('')
    else:
        axes[1].scatter(loan_df['log_loan_amount'] if 'log_loan_amount' in loan_df else np.arange(len(loan_y)), loan_y, s=2)
        axes[1].set(title='Interest rate vs loan size / observation', ylabel='Interest rate')
    plt.tight_layout()
    plt.show()

    # Pipeline 1: One-hot encoded regularized linear model, tune alpha.
    ridge = Pipeline([
        ('preprocess', preprocessing(loan_X, loan_features, scale=True)),
        ('model', Ridge())
    ])
    ridge_search = GridSearchCV(ridge, {'model__alpha': [1, 10, 100]},
                                scoring=SCORING, cv=KFold(n_splits=3, shuffle=True, random_state=RANDOM_STATE), n_jobs=1, refit=True)
    ridge_search.fit(loan_X, loan_y)

    # Pipeline 2: nonlinear boosted trees with ordinal-encoded categorical predictors.
    # The ordinal categories are not interpreted as a meaningful rank by the notebook;
    # HGB models can still find thresholds, though the encoding is an approximation.
    hgb = Pipeline([
        ('preprocess', preprocessing(loan_X, loan_features, scale=False, ordinal=True)),
        ('model', HistGradientBoostingRegressor(random_state=RANDOM_STATE,
                                                max_iter=60, early_stopping=True))
    ])
    hgb_search = GridSearchCV(hgb, {
        'model__max_leaf_nodes': [15],
        'model__learning_rate': [0.1],
        'model__l2_regularization': [0.0]
    }, scoring=SCORING, cv=KFold(n_splits=3, shuffle=True, random_state=RANDOM_STATE), n_jobs=1, refit=True)
    hgb_search.fit(loan_X, loan_y)

    loan_results = pd.DataFrame([

Reproducibility note¶

Uses the full Ames Housing dataset and a reproducible sample of 600 Lending Club loans (random seed 42) with five-fold CV for Ames and three-fold CV for Lending Club. All preprocessing occurs inside pipelines. Five-fold CV RMSE is an estimated out-of-sample score, and selecting the winner using these same folds may introduce optimism. The lending data include only originated loans.
